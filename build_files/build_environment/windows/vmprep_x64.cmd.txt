@echo off
REM  ###########################################################################
REM  #
REM  # Note: This file has a .txt extension so it cannot be directly run from a
REM  # blender source clone, it is meant to run on a clean windows install and
REM  # nothing else. DO NOT REMOVE THE .TXT extension. See instructions below on
REM  # how to execute this script.
REM  #
REM  ###########################################################################

REM Arguments (any order):
REM   unattended              - skip prompts and run silently
REM   <branchname>            - blender branch to use (default: main)
REM Example: vmprep.cmd unattended blender-v5.2-release
set UNATTENDED=0
set BRANCH=main

REM THIS IS ONLY USED FOR WIP TESTING THIS BRANCH
set TEMP_WIP_BRANCH=bartvdbraak/blender/raw/branch/windows-vmprep-improvements

:parse_args
if "%1"=="unattended" (
    set UNATTENDED=1
    shift
    goto :parse_args
)
if not "%1"=="" (
    set BRANCH=%1
    shift
    goto :parse_args
)

echo ###########################################################################
echo #
echo # This sets up a Virtual Machine for building the blender dependencies, No
echo # additional software nor a clone of the blender repository is required
echo # before running this script. A Clean installation of Windows will suffice.
echo # Obtain this script on the VM by running from an *administrator* command
echo # prompt :
echo #
echo # curl https://projects.blender.org/blender/blender/raw/branch/main/build_files/build_environment/windows/vmprep_x64.cmd.txt -o %USERPROFILE%/vmprep.cmd
echo #
echo # Then run it using %USERPROFILE%/vmprep.cmd
echo #
echo # This will install :
echo # - Visual Studio 2022 17.14.14 Buildtools with the required workloads in
echo #   the C:\vs2022bt\ folder
echo # - Meson 1.9.1
echo # - Visual Studio 2010 X64 redist (required for the yaml binary)
echo # - Git 2.38.0
echo # - CMake 3.31.7
echo # - CUDA 12.8.0
echo # - HIP 7.1.51803
echo #
echo # Additionally this will also create and populate the following folders :
echo #
echo # - C:\install\    - This is a temporary folder with installation files
echo #                    needed by this script.
echo # - C:\t\          - This is a temporary folder for the python dependencies
echo #                    to build in, as they would have build issues if the
echo #                    default user profile path is used as it is too long.
echo # - C:\blendergit\blender
echo #                  - This is the blender source repository
echo # - C:\db          - This is the build folder for the dependencies
echo #                    it has a short name to sidestep any long path issues.
echo ###########################################################################
echo #
echo # Since this will download and install a whole bunch of stuff the following
echo # disclaimer is seemingly required:
echo #
echo # ******* WARNING *******
echo # ******* WARNING *******
echo #
echo # DO NOT RUN THIS ON ANYTHING BUT A CLEAN WINDOWS INSTALL, IF YOU DO, AND IT
echo # MESSES UP YOUR WORKSTATION, THAT'S ON YOU...
echo #
echo # ******* WARNING *******
echo # ******* WARNING *******
echo #
echo # hit ctrl-break right now to abort.. last chance
echo #
if %UNATTENDED%==0 pause

echo Create install folder
mkdir C:\install

echo Create shortly named temp folder
mkdir C:\t

echo Obtaining Visual Studio 2022 Buildtools installer
curl -s https://download.visualstudio.microsoft.com/download/pr/c2e2845d-bdff-44fc-ac00-3d488e9f5675/abdb87ff12fe1885ccc4964b4ab51da566549fd2096d7ceb1fdea907372f21d7/vs_BuildTools.exe -o C:\install\vs_BuildTools.exe
echo Obtaining Visual Studio 2022 Buildtools installer configuration
curl -s https://projects.blender.org/%TEMP_WIP_BRANCH%/build_files/build_environment/windows/vsconfig_x64 -o C:\install\.vsconfig
echo Installing Visual Studio 2022 Buildtools installer
C:\install\vs_BuildTools.exe --wait --quiet --norestart --installPath C:\vs2022bt\ --config C:\install\.vsconfig

echo Obtaining Git
curl -s -L https://github.com/git-for-windows/git/releases/download/v2.38.0.windows.1/Git-2.38.0-64-bit.exe -o C:\install\Git-2.38.0-64-bit.exe
echo Installing Git
start /wait C:\install\Git-2.38.0-64-bit.exe /verysilent /norestart
echo Adding Git to PATH
set PATH=%PATH%;C:\Program Files\Git\cmd

echo Obtaining CMake
curl -s -L https://github.com/Kitware/CMake/releases/download/v3.31.7/cmake-3.31.7-windows-x86_64.msi -o C:\install\cmake-3.31.7-windows-x86_64.msi
echo Installing CMake
start /wait msiexec /quiet /norestart /i C:\install\cmake-3.31.7-windows-x86_64.msi ADD_CMAKE_TO_PATH="System"
echo Adding CMake to PATH
set PATH=%PATH%;C:\Program Files\CMake\bin

echo Obtaining Meson
curl -s -L https://github.com/mesonbuild/meson/releases/download/1.9.1/meson-1.9.1-64.msi -o C:\install\meson-1.9.1-64.msi
echo Installing Meson
start /wait msiexec /quiet /norestart /i C:\install\meson-1.9.1-64.msi
echo Adding Meson to PATH
set PATH=%PATH%;C:\Program Files\Meson\

echo Obtaining Visual studio 2010 x64 redist
curl -s https://download.microsoft.com/download/3/2/2/3224B87F-CFA0-4E70-BDA3-3DE650EFEBA5/vcredist_x64.exe -o C:\install\vcredist_x64.exe
echo Installing Visual studio 2010 x64 redist
start /wait C:\install\vcredist_x64.exe /passive /norestart

echo Obtaining CUDA 12.8.0
curl -s https://developer.download.nvidia.com/compute/cuda/12.8.0/local_installers/cuda_12.8.0_571.96_windows.exe -o C:\install\cuda_12.8.0_571.96_windows.exe
echo Installing CUDA 12.8.0
start /wait C:\install\cuda_12.8.0_571.96_windows.exe -s -n nvcc_12.8 cudart_12.8 nvrtc_12.8 nvrtc_dev_12.8 nvjitlink_12.8 nvtx_12.8 thrust_12.8 curand_12.8 curand_dev_12.8
echo Moving CUDA 12.8.0 install to C:\tools\cuda\12.8.0\
robocopy "C:\Program Files\NVIDIA GPU Computing Toolkit\CUDA\v12.8" "C:\tools\cuda\12.8.0" /E /MOVE /NJH /NJS /NP /NFL /NDL
echo Setting system wide values for CUDA_PATH and CUDA_PATH_V12_8
setx CUDA_PATH "C:\tools\cuda\12.8.0" /M > nul
setx CUDA_PATH_V12_8 "C:\tools\cuda\12.8.0" /M > nul

echo Obtaining 7zr
curl -s -L https://github.com/ip7z/7zip/releases/download/26.02/7zr.exe -o C:\install\7zr.exe
echo Adding 7zr to PATH
SET PATH=%PATH%;C:\install

echo Obtaining HIP SDK 7.1.51803-d3a86bd04
curl -s https://download.amd.com/developer/eula/rocm-hub/AMD-Software-PRO-Edition-26.Q1-Win11-For-HIP.exe -o C:\install\AMD-Software-PRO-Edition-26.Q1-Win11-For-HIP.exe
echo Extracting HIP SDK 7.1.51803-d3a86bd04 installer
7zr.exe x C:\install\AMD-Software-PRO-Edition-26.Q1-Win11-For-HIP.exe -oC:\install\ -y > nul
mkdir C:\tools\rocm
echo Installing HIP SDK 7.1.51803-d3a86bd04
start /wait msiexec /quiet /norestart /i C:\install\Packages\Apps\ROCmSDKPackages\SDKCore\ROCm_SDK_Core.msi INSTALLDIR="C:\tools\rocm"
echo Setting system wide values for HIP_PATH and HIP_PATH_71
setx HIP_PATH "C:\tools\rocm\7.1" /M > nul
setx HIP_PATH_71 "C:\tools\rocm\7.1" /M > nul

echo Cloning Blender repository
mkdir C:\blendergit
cd C:\blendergit
git clone --quiet --branch %BRANCH% https://projects.blender.org/blender/blender.git

mkdir C:\db
echo Obtaining vmbuild.cmd
curl -s https://projects.blender.org/%TEMP_WIP_BRANCH%/build_files/build_environment/windows/vmbuild_x64.cmd.txt -o C:\db\vmbuild.cmd
echo Obtaining nuke.cmd
curl -s https://projects.blender.org/blender/blender/raw/branch/%BRANCH%/build_files/build_environment/windows/nuke.cmd -o C:\db\nuke.cmd
echo Obtaining nuke_python.cmd
curl -s https://projects.blender.org/blender/blender/raw/branch/%BRANCH%/build_files/build_environment/windows/nuke_python.cmd -o C:\db\nuke_python.cmd
echo Obtaining nuke_embree.cmd
curl -s https://projects.blender.org/blender/blender/raw/branch/%BRANCH%/build_files/build_environment/windows/nuke_embree.cmd -o C:\db\nuke_embree.cmd
echo Obtaining nuke_shaderc.cmd
curl -s https://projects.blender.org/blender/blender/raw/branch/%BRANCH%/build_files/build_environment/windows/nuke_shaderc.cmd -o C:\db\nuke_shaderc.cmd

echo *******************************************************************
echo ** Run "call C:\vs2022bt\VC\Auxiliary\Build\vcvars64.bat" to     **
echo ** start "x64 Native Tools Command Prompt for VS 2022" and       **
echo ** run "cd C:\db && vmbuild.cmd" to start a build.               **
echo *******************************************************************
